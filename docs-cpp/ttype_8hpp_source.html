100 ref_ptr<TType<T>> t(new TType<T>());
101 t->m_name = std::move(name);

103 t->m_cpp = std::move(cpp);
104 t->m_desc = std::move(desc);
105 t->m_size = static_cast<int>(sizeof(T));
106 t->m_id = id;
107 return t.template as<Type>();
